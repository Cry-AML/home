{{/* Render an author list with the PI emphasised. The data stores plain text ("…, Man CH#, …"), so the split happens here: every segment containing the PI's initials is wrapped in , and a trailing role marker (# or *) stays outside the emphasis. The regex is bounded so "Leung AYH" can never match "Man CH". */}} {{- $authors := .authors -}} {{- $parts := split $authors ", " -}} {{- range $i, $a := $parts -}} {{- if $i }}, {{ end -}} {{- $isPi := findRE `^\s*Man\s*C\.?H\.?` $a -}} {{- $marks := findRE `[#*]+$` $a -}} {{- $name := replaceRE `[#*]+$` "" $a -}} {{- if $isPi -}} {{ $name | strings.TrimSpace }}{{ delimit $marks "" }} {{- else -}} {{ $a }} {{- end -}} {{- end -}}
